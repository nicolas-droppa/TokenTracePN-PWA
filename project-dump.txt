import React from 'react';
import { useTheme } from '../../../store/useSettingsStore';

const CELL = 'min-w-[44px] whitespace-nowrap px-2.5 py-2 text-center';
const STICKY_TOP = 'sticky top-0 z-10';
const STICKY_LEFT = 'sticky left-0 z-10';
const STICKY_CORNER = 'sticky left-0 top-0 z-20';

/**
 * Matrix with sticky headers
 * @param {Object} props
 * @param {string} props.title
 * @param {{ id: string, label: string }[]} props.rows - Row headers, in matrix order.
 * @param {{ id: string, label: string }[]} props.columns - Column headers, in matrix order.
 * @param {number[][]} props.matrix
 * @param {{ ref: Function, onScroll: Function }} [props.scroll] - Optional synced scrolling.
 */
export const MatrixTable = ({ title, rows, columns, matrix, scroll }) => {
    const theme = useTheme();
    const headerStyle = { backgroundColor: theme.sidebar.bg, color: theme.text.label };

    return (
        <section
            className="flex min-h-0 min-w-0 flex-1 flex-col overflow-hidden rounded-md"
            style={{ backgroundColor: theme.sidebar.bg }}
        >
            <h3
                className="px-2.5 py-2 text-xs font-bold uppercase tracking-wide"
                style={{ color: theme.text.label }}
            >
                {title}
            </h3>

            <div ref={scroll?.ref} onScroll={scroll?.onScroll} className="min-h-0 flex-1 overflow-auto">
                <table className="min-w-max border-collapse text-xs" style={{ color: theme.text.code }}>
                    <thead>
                        <tr>
                            <th className={`${CELL} ${STICKY_CORNER}`} style={headerStyle} />
                            {columns.map((column) => (
                                <th
                                    key={column.id}
                                    scope="col"
                                    title={column.id}
                                    className={`${CELL} ${STICKY_TOP}`}
                                    style={headerStyle}
                                >
                                    {column.label}
                                </th>
                            ))}
                        </tr>
                    </thead>

                    <tbody>
                        {matrix.map((values, rowIndex) => {
                            const row = rows[rowIndex];

                            return (
                                <tr key={row.id}>
                                    <th
                                        scope="row"
                                        title={row.id}
                                        className={`${CELL} ${STICKY_LEFT} text-left`}
                                        style={headerStyle}
                                    >
                                        {row.label}
                                    </th>
                                    {values.map((value, columnIndex) => (
                                        <ValueCell key={columns[columnIndex].id} value={value} theme={theme} />
                                    ))}
                                </tr>
                            );
                        })}
                    </tbody>
                </table>
            </div>
        </section>
    );
};

/**
 * A single matrix value.
 */
const ValueCell = ({ value, theme }) => {
    const isZero = value === 0;

    return (
        <td
            className={`${CELL} font-mono`}
            style={{
                color: isZero ? theme.disabled.text : theme.text.code,
                backgroundColor: isZero ? 'transparent' : `${theme.text.label}14`,
            }}
        >
            {value}
        </td>
    );
};

import React from 'react';
import { usePetriStore } from '../../store/usePetriStore';
import { useTheme } from '../../store/useSettingsStore';
import { buildIOMatrices } from '../../core/analysis/matrices';
import { displayName } from '../../core/models';
import { useSyncedScroll } from '../../hooks/useSyncedScroll';
import { MatrixTable } from './components/MatrixTable';

/**
 * Input and output matrices of the current net, recomputed only on change.
 */
export const MatricesView = () => {
    const theme = useTheme();
    const places = usePetriStore((s) => s.places);
    const transitions = usePetriStore((s) => s.transitions);
    const arcs = usePetriStore((s) => s.arcs);
    const bindScroll = useSyncedScroll();

    const matrices = React.useMemo(
        () => buildIOMatrices({ places, transitions, arcs }),
        [places, transitions, arcs]
    );

    const rows = React.useMemo(() => places.map(toHeader), [places]);
    const columns = React.useMemo(() => transitions.map(toHeader), [transitions]);

    const hasIgnoredArcs = arcs.some((arc) => arc.arcType !== 'regular');

    if (places.length === 0 || transitions.length === 0) {
        return (
            <p className="p-3 text-xs" style={{ color: theme.disabled.text }}>
                Add places and transitions to see the matrices.
            </p>
        );
    }

    return (
        <div className="flex h-full min-h-0 flex-col gap-2 p-2" style={{ backgroundColor: theme.sidebar.inputBg }}>
            {hasIgnoredArcs && (
                <p className="px-1 text-xs" style={{ color: theme.disabled.text }}>
                    Read, inhibitor and reset arcs are not shown in the matrices.
                </p>
            )}

            <div className="flex min-h-0 flex-1 gap-2">
                <MatrixTable
                    title="Pre (I)"
                    rows={rows}
                    columns={columns}
                    matrix={matrices.input}
                    scroll={bindScroll(0)}
                />
                <MatrixTable
                    title="Post (O)"
                    rows={rows}
                    columns={columns}
                    matrix={matrices.output}
                    scroll={bindScroll(1)}
                />
            </div>
        </div>
    );

import React from 'react';
import { usePetriStore } from '../../../store/usePetriStore';
import { useSettingsStore } from '../../../store/useSettingsStore';
import { THEMES } from '../../../theme';
import { ARC_CONFIG } from '../../../constants/layout';

/**
 * ArcEdge component for rendering an arc connection between nodes with optional weight display.
 *
 * @param {Object} props - Component properties.
 * @param {Object} props.arc - The arc object data (contains id and weight).
 * @param {Object} props.sourceNode - The source node object (Place or Transition).
 * @param {Object} props.targetNode - The target node object (Place or Transition).
 * @returns {JSX.Element|null} SVG group element or null if nodes are missing.
 */
export const ArcEdge = ({ arc, sourceNode, targetNode }) => {
  const activeThemeKey = useSettingsStore((state) => state.activeTheme);
  const theme = THEMES[activeThemeKey] || THEMES.dark;
  const select = usePetriStore((state) => state.select);
  const isSelected = usePetriStore((state) => state.selectedId === arc.id);

  if (!sourceNode || !targetNode) return null;

  const markerId = `arrow-${arc.id}`;

  const { midX, midY } = calculateMidpoint(sourceNode, targetNode);

  const strokeColor = isSelected ? theme.place.stroke : theme.arc.stroke;
  const strokeWidth = isSelected ? ARC_CONFIG.STROKE_WIDTH_PX + 1 : ARC_CONFIG.STROKE_WIDTH_PX;

  const handleClick = (e) => {
    e.stopPropagation();
    select(arc.id);
  };

  return (
    <g
      className="arc-edge"
      data-element-id={arc.id}
      data-element-type="arc"
    >
      <defs>
        {/* Arrowhead Marker */}
        <marker
          id={markerId}
          viewBox="0 0 10 10"
          refX="28"
          refY="5"
          markerWidth="6"
          markerHeight="6"
          orient="auto-start-reverse"
        >
          <path d="M 0 0 L 10 5 L 0 10 z" fill={strokeColor} />
        </marker>
      </defs>

      {/* Hit Area (widens area) */}
      <line
        x1={sourceNode.x}
        y1={sourceNode.y}
        x2={targetNode.x}
        y2={targetNode.y}
        stroke="transparent"
        strokeWidth={14}
        pointerEvents="stroke"
        className="cursor-pointer"
        onClick={handleClick}
      />

      {/* Connection Line */}
      <line
        x1={sourceNode.x}
        y1={sourceNode.y}
        x2={targetNode.x}
        y2={targetNode.y}
        stroke={strokeColor}
        strokeWidth={strokeWidth}
        markerEnd={`url(#${markerId})`}
        pointerEvents="none"
      />

      {/* Arc Weight Label (renders only if weight > 1) */}
      {arc.weight > 1 && (
        <>
          <circle
            cx={midX}
            cy={midY - 10}
            r="9"
            fill={theme.bg}
            pointerEvents="none"
          />
          <text
            x={midX}
            y={midY - 6}
            fill={strokeColor}
            fontSize="13"
            fontWeight="bold"
            textAnchor="middle"
            className="select-none"
            pointerEvents="none"
          >
            {arc.weight}
          </text>
        </>
      )}
    </g>
  );
};

// private

/**
 * Calculates the center midpoint between source and target nodes.
 *
 * @param {Object} source - Source node coordinates {x, y}.
 * @param {Object} target - Target node coordinates {x, y}.
 * @returns {{ midX: number, midY: number }}
 */
const calculateMidpoint = (source, target) => {
  return {
    midX: (source.x + target.x) / 2,
    midY: (source.y + target.y) / 2,
  };
};

import React from 'react';
import { useTheme } from '../../../store/useSettingsStore';
import { CANVAS_CONFIG } from '../../../constants/layout';
import { range } from '../../../utils/array';
import { getVisibleCells, canvasToScreen } from '../../../hooks/canvas/helpers/canvasPanZoom';
import { cellToCanvas } from '../../../utils/grid';
import { getLabelStep } from '../../../hooks/canvas/helpers/canvasRulers';

const rulerThickness = CANVAS_CONFIG.RULER_THICKNESS_PX;
const fontSize = 11;

export const CanvasRulers = ({ width = 0, height = 0, pan, zoom }) => {
    const theme = useTheme();

    if (width === 0 || height === 0) return null;

    const { minX, maxX, minY, maxY } = getVisibleCells(width, height, pan, zoom);
    const step = getLabelStep(zoom);
    const columns = range(minX, maxX).filter((col) => col % step === 0);
    const rows = range(minY, maxY).filter((row) => row % step === 0);

    return (
        <g>
            <Ruler orientation="horizontal" height={height} width={width} theme={theme}/>
            <Ruler orientation="vertical" height={height} width={width} theme={theme}/>

            {columns.map((col) => (
                <text
                    key={`col-${col}`}
                    x={ columnScreenX(col, pan, zoom) }
                    y={ height - rulerThickness / 2 }
                    textAnchor="middle"
                    dominantBaseline="middle"
                    fontSize={ fontSize }
                    fill={ col === 0 ? theme.rulers.accent : col % 5 === 0 ? theme.rulers.text : theme.rulers.tick }
                >
                    {col}
                </text>
            ))}

            {rows.map((row) => (
                <text
                    key={`row-${row}`}
                    x={ width - rulerThickness / 2 }
                    y={ rowScreenY(row, pan, zoom) }
                    textAnchor="middle"
                    dominantBaseline="middle"
                    fontSize={ fontSize }
                    fill={ row === 0 ? theme.rulers.accent : row % 5 === 0 ? theme.rulers.text : theme.rulers.tick }
                >
                    {row}
                </text>
            ))}

            <RulerConnector height={height} width={width} theme={theme}/>
        </g>
    );
};

const Ruler = ({ orientation, height, width, theme }) => {
    const isHorizontal = orientation === 'horizontal';

    return (
        <rect
            x={ isHorizontal ? 0 : width - rulerThickness }
            y={ isHorizontal ? height - rulerThickness : 0 }
            height={ isHorizontal ? rulerThickness : height - rulerThickness }
            width={ isHorizontal ? width - rulerThickness : rulerThickness }
            fill={`${theme.rulers.bg}e6`}
        />
    );
};

const RulerConnector = ({ height, width, theme }) => {
    return (
        <rect
            x={ width - rulerThickness }
            y={ height - rulerThickness }
            height={ rulerThickness }
            width={ rulerThickness }
            fill={`${theme.rulers.bg}e6`}
        />
    );
};

const columnScreenX = (col, pan, zoom) => {
    return canvasToScreen(cellToCanvas({ x: col, y: 0 }), pan, zoom).x
}

const rowScreenY = (row, pan, zoom) => {
    return canvasToScreen(cellToCanvas({ x: 0, y: row }), pan, zoom).y
}

===== src/components/canvas/components/TransitionNode.jsx
import React from 'react';
import { useSettingsStore } from '../../../store/useSettingsStore';
import { useNodeConnecting } from '../../../hooks/canvas/useNodeConnecting';
import { displayName } from '../../../core/models';
import { THEMES } from '../../../theme';
import { NODE_CONFIG } from '../../../constants/layout';

/**
 * Renders a single transition. In run mode an enabled transition is filled
 * and clickable; firing it flashes briefly as feedback.
 *
 * @param {Object} props
 * @param {Object} props.transition - The transition object.
 * @param {boolean} props.isRunning - Whether simulation is active.
 * @param {boolean} props.isEnabled - Whether transition can fire.
 * @param {Function} props.onFire - Fires the transition.
 * @param {Function} props.onMouseDown - Drag start handler.
 */
export const TransitionNode = ({
  transition,
  isRunning = false,
  isEnabled = false,
  onFire,
  onMouseDown,
}) => {
  const activeThemeKey = useSettingsStore((state) => state.activeTheme);
  const theme = THEMES[activeThemeKey] || THEMES.dark;

  const [isFlashing, setIsFlashing] = React.useState(false);

  const { isConnectingSource, isInvalidTarget, containerProps } = useNodeConnecting(transition, onMouseDown);

  const size = NODE_CONFIG.TRANSITION_SIZE_PX;

  const isFireable = isRunning && isEnabled;
  const isDisabled = isInvalidTarget || (isRunning && !isEnabled);

  const strokeColor = isDisabled ? theme.disabled.stroke : theme.transition.stroke;
  const labelColor = isDisabled ? theme.disabled.text : theme.text.label;

  const handleClick = (e) => {
    if (!isRunning) {
      containerProps.onClick(e);
      return;
    }
    e.stopPropagation();
    if (!isEnabled) return;

    setIsFlashing(true);
    setTimeout(() => setIsFlashing(false), 140);
    onFire?.();
  };

  const cursorClass = isRunning
    ? isEnabled
      ? 'cursor-pointer'
      : 'cursor-not-allowed'
    : undefined;

  return (
    <g
      transform={`translate(${transition.x}, ${transition.y})`}
      {...containerProps}
      onClick={handleClick}
      className={cursorClass ?? containerProps.className}
    >
      {/* Connecting Indicator */}
      {isConnectingSource && (
        <rect
          x={-size / 2 - 6}
          y={-size / 2 - 6}
          width={size + 12}
          height={size + 12}
          fill="none"
          stroke={theme.arc.stroke}
          strokeWidth="2"
          strokeDasharray="4 4"
          rx={NODE_CONFIG.TRANSITION_CORNER_RADIUS_PX + 2}
          pointerEvents="none"
        />
      )}

      {/* Shape */}
      <rect
        x={-size / 2}
        y={-size / 2}
        width={size}
        height={size}
        fill={theme.transition.fill}
        stroke={isFlashing ? theme.text.token : strokeColor}
        strokeWidth={NODE_CONFIG.STROKE_WIDTH_PX}
        rx={NODE_CONFIG.TRANSITION_CORNER_RADIUS_PX}
        transform={`scale(${isFlashing ? 0.9 : 1})`}
        style={{
          transformOrigin: 'center',
          transformBox: 'fill-box',
          transition: isFlashing
            ? 'transform 100ms ease-out'
            : 'transform 100ms ease-out, stroke 120ms ease-out',
        }}
      />

      {/* Label */}
      <text
        textAnchor="middle"
        y={size / 2 + 18}
        fill={labelColor}
        className="text-xs font-medium select-none"
        pointerEvents="none"
      >
        {displayName(transition)}
      </text>
    </g>
  );
};

const getSeverityMeta = (severity, theme) => {
    const palette = theme?.diagnostics ?? {};
    const bySeverity = {
        error: { color: palette.error, label: 'Error' },
        warning: { color: palette.warning, label: 'Warning' },

export const CodeCheckerPanel = ({ errors = [], theme = {}, code = '', language = 'dpn' }) => {

    const counts = {

                                    </div>

import React, { useEffect, useRef } from 'react';
import { DockviewReact, themeDark, themeLight } from 'dockview-react';
import { useSettingsStore, useTheme } from '../../store/useSettingsStore';
import { useWorkspaceStore } from '../../store/useWorkspaceStore';
import { LAYOUT_VERSION } from '../../constants/workspace';
import { applyDefaultLayout } from '../../utils/dockLayout';
import { PANEL_COMPONENTS } from './panelComponents';
import './dockviewTheme.css';

const SAVE_DELAY_MS = 300;

/** App theme -> CSS variables read by dockviewTheme.css. */
const dockThemeVars = (theme) => ({
    '--tt-panel-bg': theme.sidebar.bg,
    '--tt-tabbar-bg': theme.sidebar.bg,
    '--tt-tab-active-bg': theme.bg,
    '--tt-border': theme.sidebar.border,
    '--tt-text-strong': theme.text.token,
    '--tt-text': theme.text.label,
    '--tt-text-muted': theme.disabled.text,
    '--tt-accent': theme.place.stroke,
    '--tt-accent-soft': `${theme.place.stroke}33`,
});

/** Restores the saved layout. */
const restoreLayout = (api) => {
    const saved = useSettingsStore.getState().workspaceLayout;
    if (saved?.version !== LAYOUT_VERSION) return false;

    try {
        api.fromJSON(saved.layout);
        return api.panels.length > 0;
    } catch {
        return false;
    }
};

/**
 * Desktop/tablet workspace.
 */
export const DockWorkspace = () => {
    const theme = useTheme();
    const activeTheme = useSettingsStore((s) => s.activeTheme);
    const setDockApi = useWorkspaceStore((s) => s.setDockApi);
    const cleanupRef = useRef(null);

    const handleReady = ({ api }) => {
        cleanupRef.current?.();

        if (!restoreLayout(api)) applyDefaultLayout(api);
        setDockApi(api);

        // Save the layout after the user stops interacting
        let timer;
        const subscription = api.onDidLayoutChange(() => {
            clearTimeout(timer);
            timer = setTimeout(() => {
                useSettingsStore.getState().setWorkspaceLayout({ version: LAYOUT_VERSION, layout: api.toJSON() });
            }, SAVE_DELAY_MS);
        });

        cleanupRef.current = () => {
            clearTimeout(timer);
            subscription.dispose();
        };
    };

    useEffect(
        () => () => {
            cleanupRef.current?.();
            setDockApi(null);
        },
        [setDockApi]
    );

    return (
        <div className="tt-dock h-full w-full" style={dockThemeVars(theme)}>
            <DockviewReact
                components={PANEL_COMPONENTS}
                onReady={handleReady}
                theme={activeTheme === 'light' ? themeLight : themeDark}
            />
        </div>
    );

===== src/core/analysis/matrices.js
import { zeros } from '../../utils/array';

/**
 * Builds the input (pre) and output (post) matrices of the net.
 * Non regular arcs ignored.
 * @param {Object} net - Petri net.
 * @returns {{ places: string[], transitions: string[], input: number[][], output: number[][] }}
 */
export function buildIOMatrices(net) {
    const input = zeros(net.places.length, net.transitions.length);
    const output = zeros(net.places.length, net.transitions.length);

    const mapPlaceToRowIndex = new Map();
    const places = [];
    for (const place of net.places) {
        places.push(place.id);
        mapPlaceToRowIndex.set(place.id, mapPlaceToRowIndex.size);
    }

    const mapTransitionToColumnIndex = new Map();
    const transitions = [];
    for (const transition of net.transitions) {
        transitions.push(transition.id);
        mapTransitionToColumnIndex.set(transition.id, mapTransitionToColumnIndex.size);
    }

    for (const arc of net.arcs) {
        if (arc.arcType !== "regular") continue;

        // place -> transition
        const inputRow = mapPlaceToRowIndex.get(arc.source);
        const inputColumn = mapTransitionToColumnIndex.get(arc.target);
        if (inputRow !== undefined && inputColumn !== undefined) {
            input[inputRow][inputColumn] = arc.weight;
        }

        // transition -> place
        const outputRow = mapPlaceToRowIndex.get(arc.target);
        const outputColumn = mapTransitionToColumnIndex.get(arc.source);
        if (outputRow !== undefined && outputColumn !== undefined) {
            output[outputRow][outputColumn] = arc.weight;
        }
    }

    return {
        places,
        transitions,
        input,
        output,
    };
}

/**
 * Petri net model - single source of truth
 *
 * @typedef {Object} Place
 * @property {string} id
 * @property {'place'} type
 * @property {number} x
 * @property {number} y
 * @property {string} label
 * @property {number} tokens
 *
 * @typedef {Object} Transition
 * @property {string} id
 * @property {'transition'} type
 * @property {number} x
 * @property {number} y
 * @property {string} label
 *
 * @typedef {Object} Arc
 * @property {string} id
 * @property {'arc'} type
 * @property {string} source
 * @property {string} target
 * @property {number} weight
 * @property {'regular'|'read'|'inhibitor'|'reset'} arcType
 *
 * @typedef {Object} Net
 * @property {Place[]} places
 * @property {Transition[]} transitions
 * @property {Arc[]} arcs
 */

export const ARC_TYPES = ['regular', 'read', 'inhibitor', 'reset'];

export const createPlace = ({ id, x = 0, y = 0, label = '', tokens = 0 }) => ({
    id,
    type: 'place',
    x,
    y,
    label,
    tokens: Math.max(0, Math.floor(Number(tokens) || 0)),
});

export const createTransition = ({ id, x = 0, y = 0, label = '' }) => ({
    id,
    type: 'transition',
    x,
    y,
    label,
});

export const createArc = ({ source, target, weight = 1, arcType = 'regular' }) => ({
    id: arcId(source, target),
    type: 'arc',
    source,
    target,
    weight: Math.max(1, Math.floor(Number(weight) || 1)),
    arcType: ARC_TYPES.includes(arcType) ? arcType : 'regular',
});

export const arcId = (source, target) => `${source}->${target}`;

/**
 * First free node name with the given prefix: p1, p2, ... .
 * @param {'p'|'t'} prefix
 * @param {{ places: Place[], transitions: Transition[] }} net
 * @returns {string}
 */
export const nextNodeId = (prefix, { places, transitions }) => {
    const used = new Set([...places, ...transitions].map((node) => node.id));
    let n = 1;
    while (used.has(`${prefix}${n}`)) n++;
    return `${prefix}${n}`;
};

export const displayName = (node) => node.label || node.id;

/**
 * Finds a place, transition or arc by id.
 * @param {{ places: Place[], transitions: Transition[], arcs: Arc[] }} net - The net (or the store state).
 * @param {string|null} id
 * @returns {Place|Transition|Arc|null}
 */
export const findElement = (net, id) => {
    if (id == null) return null;
    return (
        net.places.find((el) => el.id === id) ??
        net.transitions.find((el) => el.id === id) ??
        net.arcs.find((el) => el.id === id) ??
        null
    );
};

===== src/core/netCommands.js
import { ATTRIBUTE_SCHEMA, PLACE_TO_TRANSITION_ONLY } from '../constants/dpn.js';
import { createPlace, createTransition, createArc, arcId, findElement } from './models.js';
import { matchesSchema, modelField } from './parsers/dpnSchema.js';
import { isValidArc } from './petriEngine.js';

/**
 * Commands made from users in UI, or from code in the editor.
 *
 * @typedef {(
 *   | { type: 'addPlace', id: string, x: number, y: number }
 *   | { type: 'addTransition', id: string, x: number, y: number }
 *   | { type: 'addArc', source: string, target: string, arcType?: string }
 *   | { type: 'moveNode', id: string, x: number, y: number }
 *   | { type: 'setAttribute', id: string, key: string, value: * }
 *   | { type: 'deleteElement', id: string }
 * )} NetCommand
 */

/**
 * Applies a command to the net.
 * @param {import('./models.js').Net} net
 * @param {NetCommand} command
 * @returns {import('./models.js').Net}
 */
export function applyCommand(net, command) {
    switch (command.type) {
        case 'addPlace':
            return addNode(net, 'places', createPlace({ ...command, x: round(command.x), y: round(command.y) }));
        case 'addTransition':
            return addNode(net, 'transitions', createTransition({ ...command, x: round(command.x), y: round(command.y) }));
        case 'addArc':
            return addArc(net, command);
        case 'moveNode':
            return updateNode(net, command.id, { x: round(command.x), y: round(command.y) });
        case 'setAttribute':
            return setAttribute(net, command);
        case 'deleteElement':
            return deleteElement(net, command.id);
        default:
            return net;
    }
}

// private

const round = (value) => Math.round(value);

function addNode(net, listName, node) {
    if (findElement(net, node.id)) return net; // name already taken
    return { ...net, [listName]: [...net[listName], node] };
}

function addArc(net, { source, target, arcType }) {
    if (!isValidArc(source, target, net.places, net.transitions)) return net;
    if (findElement(net, arcId(source, target))) return net; // one arc per pair of nodes

    return { ...net, arcs: [...net.arcs, createArc({ source, target, arcType })] };
}

function updateNode(net, id, changes) {
    const update = (node) => (node.id === id ? { ...node, ...changes } : node);
    const element = findElement(net, id);
    if (!element || element.type === 'arc') return net;

    return { ...net, places: net.places.map(update), transitions: net.transitions.map(update) };
}

function setAttribute(net, { id, key, value }) {
    const element = findElement(net, id);
    if (!element) return net;

    const expected = ATTRIBUTE_SCHEMA[element.type]?.[key];
    if (expected === undefined || !matchesSchema(value, expected)) return net;

    // inhibitor / read / reset goes only from place to transition
    if (element.type === 'arc' && key === 'type' && PLACE_TO_TRANSITION_ONLY.has(value)) {
        if (findElement(net, element.source)?.type !== 'place') return net;
    }

    const changes = { [modelField(element.type, key)]: value };

    if (element.type !== 'arc') return updateNode(net, id, changes);
    return { ...net, arcs: net.arcs.map((arc) => (arc.id === id ? { ...arc, ...changes } : arc)) };
}

function deleteElement(net, id) {
    const element = findElement(net, id);
    if (!element) return net;

    if (element.type === 'arc') return { ...net, arcs: net.arcs.filter((arc) => arc.id !== id) };

    return {
        places: net.places.filter((node) => node.id !== id),
        transitions: net.transitions.filter((node) => node.id !== id),
        arcs: net.arcs.filter((arc) => arc.source !== id && arc.target !== id),
    };
}

===== src/core/parsers/dpnBuilder.js
import { ATTRIBUTE_SCHEMA } from '../../constants/dpn.js';
import { AUTO_LAYOUT } from '../../constants/layout.js';
import { createPlace, createTransition, createArc, arcId } from '../models.js';
import { canonicalKey, kindOf, matchesSchema, modelField } from './dpnSchema.js';

/**
 * Builds the Petri net model (see core/models.js) from a DPN AST.
 *
 * Meant to run only on code without errors. Whatever the semantic checker reports as
 * a warning is resolved the way its message says: the first declaration, the first arc
 * between two nodes and the first value of an attribute win, the rest is ignored.
 *
 * @param {Object} ast - `model` returned by dpnParser.parse().
 * @returns {import('../models.js').Net}
 */
export function buildNet(ast) {
    const statements = ast?.statements ?? [];
    const symbols = collectDeclarations(statements);
    const nodeFields = collectNodeFields(statements, symbols);

    const places = [];
    const transitions = [];
    let declarationIndex = 0;

    for (const [name, declaration] of symbols) {
        const fields = nodeFields.get(name);
        // Grid cell = order of the declaration, so moving another node never shifts this one
        const auto = autoPosition(declarationIndex++);
        const node = { ...fields, id: name, x: fields.x ?? auto.x, y: fields.y ?? auto.y };

        if (declaration.keyword === 'place') places.push(createPlace(node));
        else transitions.push(createTransition(node));
    }

    return { places, transitions, arcs: buildArcs(statements, symbols) };
}

// private

/**
 * Name -> first declaration of that name (later duplicates are errors and are ignored).
 * @returns {Map<string, Object>}
 */
function collectDeclarations(statements) {
    const symbols = new Map();

    for (const statement of statements) {
        if (statement.type === 'declaration' && !symbols.has(statement.name)) {
            symbols.set(statement.name, statement);
        }
    }

    return symbols;
}

/**
 * Name -> model fields of that node, merged from its declaration and all its
 * configurations in source order (`p1 (x: 10);` after `place p1;`).
 * @returns {Map<string, Object>}
 */
function collectNodeFields(statements, symbols) {
    const nodeFields = new Map([...symbols.keys()].map((name) => [name, {}]));

    for (const statement of statements) {
        if (statement.type === 'arc') continue;
        if (statement.type === 'declaration' && symbols.get(statement.name) !== statement) continue; // duplicate

        const fields = nodeFields.get(statement.name);
        if (!fields) continue; // configuration of an undeclared node

        addAttributes(fields, statement.attributes, kindOf(statement, symbols));
    }

    return nodeFields;
}

/**
 * Arcs between existing nodes of different kinds, one per pair of nodes.
 * @returns {import('../models.js').Arc[]}
 */
function buildArcs(statements, symbols) {
    const arcs = [];
    const seen = new Set();

    for (const statement of statements) {
        if (statement.type !== 'arc') continue;

        const from = symbols.get(statement.from);
        const to = symbols.get(statement.to);
        if (!from || !to || from.keyword === to.keyword) continue;

        const id = arcId(statement.from, statement.to);
        if (seen.has(id)) continue;
        seen.add(id);

        const fields = addAttributes({}, statement.attributes, 'arc');
        arcs.push(createArc({ ...fields, source: statement.from, target: statement.to }));
    }

    return arcs;
}

/**
 * Copies valid attributes into `fields` under their model field name.
 * A field that is already set keeps its value (first one wins).
 * @param {Object} fields - Target object, mutated and returned.
 * @param {Array<{key: string, value: *}>} attributes - From the AST.
 * @param {'place'|'transition'|'arc'} kind
 * @returns {Object} fields
 */
function addAttributes(fields, attributes, kind) {
    const schema = ATTRIBUTE_SCHEMA[kind] ?? {};

    for (const { key, value } of attributes) {
        const canonical = canonicalKey(key);
        const expected = Object.hasOwn(schema, canonical) ? schema[canonical] : undefined;
        if (expected === undefined || !matchesSchema(value, expected)) continue;

        const field = modelField(kind, canonical);
        if (!Object.hasOwn(fields, field)) fields[field] = value;
    }

    return fields;
}

/** Grid position of the n-th declared node (used only when it has no x/y in the code). */
function autoPosition(index) {
    return {
        x: AUTO_LAYOUT.ORIGIN_X + (index % AUTO_LAYOUT.COLUMNS) * AUTO_LAYOUT.SPACING,
        y: AUTO_LAYOUT.ORIGIN_Y - Math.floor(index / AUTO_LAYOUT.COLUMNS) * AUTO_LAYOUT.SPACING,

/**
 * Canvas -> code: turns a net command into a small edit of the DPN source. 
 * 
 * @param {string} code - Current DPN source.
 * @param {import('../netCommands.js').NetCommand} command
 * @returns {string|null} The new source, or null when the code has errors.
 */
export function applyCommandToCode(code, command) {
    const { model, errors } = parse(code);
    if (!model || errors.some((err) => err.severity === 'error')) return null;

    const changes = changesFor(code, model.statements, command);
    return applyChanges(code, changes);

/**
 * Short form used when attribute is written for the first time (tokens -> m).

const SHORT_KEY = Object.fromEntries(Object.entries(ATTRIBUTE_ALIASES).map(([alias, full]) => [full, alias]));

/**
 * @returns {Array<{from: number, to: number, insert: string}>}
 */
function changesFor(code, statements, command) {
    switch (command.type) {
        case 'addPlace':
        case 'addTransition': {
            const keyword = command.type === 'addPlace' ? 'place' : 'transition';
            const line = `${keyword} ${command.id} (x: ${Math.round(command.x)}, y: ${Math.round(command.y)});`;
            return [appendLine(code, statements, line, (s) => s.type !== 'arc')];
        }
        case 'addArc': {
            const type = command.arcType && command.arcType !== 'regular' ? ` (${writeAttribute('type', command.arcType)})` : '';
            return [appendLine(code, statements, `${command.source} -> ${command.target}${type};`, (s) => s.type === 'arc')];
        }
        case 'moveNode':
            return setAttributes(statements, command.id, { x: Math.round(command.x), y: Math.round(command.y) });
        case 'setAttribute':
            return setAttributes(statements, command.id, { [command.key]: command.value });
        case 'deleteElement':
            return deleteElement(code, statements, command.id);
        default:
            return [];
    }
}

/**
 * Statements that describe an element. Node knows its declaration and its arcs + Arc knows itself.
 * @param {Array<Object>} statements - AST nodes.
 * @param {string} id - Node or arc ID.
 * @returns {Array<Object>}
 */
function statementsOf(statements, id) {
    return statements.filter((s) => (s.type === 'arc' ? arcId(s.from, s.to) === id : s.name === id));
}

/**
 * Sets attributes of one element.
 * @param {Object<string, *>} values - Canonical key -> value, e.g. { x: 10, y: 20 }.
 */
function setAttributes(statements, id, values) {
    const owned = statementsOf(statements, id);
    if (owned.length === 0) return [];

    const changes = [];
    const missing = [];

    for (const [key, value] of Object.entries(values)) {
        const attr = owned.flatMap((s) => s.attributes).find((a) => canonicalKey(a.key) === key);
        if (attr) changes.push({ from: attr.valueStart, to: attr.end, insert: formatValue(value) });
        else missing.push(writeAttribute(key, value));
    }

    if (missing.length === 0) return changes;

    const target = owned.find((s) => s.type === 'declaration') ?? owned[0];
    const text = missing.join(', ');

    if (!target.attributeList) {
        return [...changes, { from: target.headerEnd, to: target.headerEnd, insert: ` (${text})` }];
    }

    const beforeParen = target.attributeList.end - 1;
    const separator = target.attributes.length > 0 ? ', ' : '';
    return [...changes, { from: beforeParen, to: beforeParen, insert: `${separator}${text}` }];
}

/**
 * Removes every statement of the element.
 * @param {string} code - Source code.
 * @param {Array<Object>} statements - AST nodes.
 * @param {string} id - ID of the element to delete.
 * @returns {Array<{from: number, to: number, insert: string}>}
 */
function deleteElement(code, statements, id) {
    const isArc = id.includes('->');
    const doomed = statements.filter((s) =>
        isArc ? s.type === 'arc' && arcId(s.from, s.to) === id : s.name === id || s.from === id || s.to === id
    );

    return mergeRanges(doomed.map((s) => removalRange(code, s)));
}

/**
 * Inserts a new line after the last statement of the same group (nodes / arcs) or at the EOF.
 */
function appendLine(code, statements, line, inGroup) {
    const last = statements.filter(inGroup).at(-1);

    if (last) {
        const at = lineEnd(code, last.end);
        return { from: at, to: at, insert: `\n${line}` };
    }

    const needsNewline = code.length > 0 && !code.endsWith('\n');
    return { from: code.length, to: code.length, insert: `${needsNewline ? '\n' : ''}${line}\n` };
}

/**
 * What to delete for statement.
 */
function removalRange(code, statement) {
    const start = code.lastIndexOf('\n', statement.start - 1) + 1;
    const end = lineEnd(code, statement.end);
    const before = code.slice(start, statement.start);
    const after = code.slice(statement.end, end).trim();

    if (before.trim() === '' && (after === '' || after.startsWith('//'))) {
        return { from: start, to: Math.min(end + 1, code.length), insert: '' };
    }

    const spaces = code.slice(statement.end).match(/^[ \t]*/)[0].length;
    return { from: statement.start, to: statement.end + spaces, insert: '' };
}

/** 
 * Joins overlapping removals so they can be applied together.
 */
function mergeRanges(ranges) {
    const sorted = [...ranges].sort((a, b) => a.from - b.from);
    const merged = [];

    for (const range of sorted) {
        const last = merged.at(-1);
        if (last && range.from <= last.to) last.to = Math.max(last.to, range.to);
        else merged.push({ ...range });
    }

    return merged;
}

/**
 * Applies non-overlapping changes, from the end so earlier offsets stay valid.
 */
function applyChanges(code, changes) {
    return [...changes]
        .sort((a, b) => b.from - a.from)
        .reduce((text, { from, to, insert }) => text.slice(0, from) + insert + text.slice(to), code);
}

/**
 * Position of the end of the line that contains index (before the newline).
 */
function lineEnd(code, index) {
    const newline = code.indexOf('\n', index);
    return newline === -1 ? code.length : newline;
}

/**
 * key: value with the short key when there is one (m: 3, l: "Idle").
 */
function writeAttribute(key, value) {
    return `${SHORT_KEY[key] ?? key}: ${formatValue(value)}`;
}

/**
 * Numbers as they are, strings quoted.
 */
function formatValue(value) {
    if (typeof value !== 'string') return String(value);
    const text = value.replace(/\n/g, ' ');
    if (!text.includes('"')) return `"${text}"`;
    if (!text.includes("'")) return `'${text}'`;
    return `"${text.replace(/"/g, '')}"`;
}

export function canParse(code) {
    try {
        const result = parse(code);
        return !(result.errors || []).some((err) => err.severity === 'error');
    } catch (err) {
        return false;
    }
}

function normalizeError(err) {
    const message = err?.message || String(err);
    const match = message.match(/line\s+(\d+),\s*col\s+(\d+)/i);

    const line = Number.isFinite(Number(err?.line)) ? Number(err.line) : match ? Number(match[1]) : undefined;
    const col = Number.isFinite(Number(err?.col)) ? Number(err.col) : match ? Number(match[2]) : undefined;

    return {
        severity: 'error',
        message,
        line,
        col,
    };
}

export function parse(code) {
    const { tokens, errors } = tokenize(code);

    if (errors.length > 0) {
        return {
            model: null,
            errors: errors.map(normalizeError)
        };
    }

    let model;
    try {
        model = parseTokens(tokens);
    } catch (err) {
        return {
            model: null,
            errors: [normalizeError(err)]
        };
    }

    return { model, errors: checkSemantics(model) };
}

export default { canParse, parse };

// private

function handleNumbers(ctx, startLine, startCol) {
    let sign = '';
    if (ctx.src[ctx.i] === '-' && /[0-9]/.test(ctx.src[ctx.i + 1])) {
        sign = '-';
        ctx.advance();
    }

    let num = '';
    while (ctx.i < ctx.src.length && /[0-9]/.test(ctx.src[ctx.i])) {
        num += ctx.src[ctx.i];
        ctx.advance();
    }

    ctx.pushToken({ type: 'NUMBER', value: Number(sign + num), line: startLine, col: startCol });
}

function handleLetters(ctx, startLine, startCol) {
    let id = '';
    while (ctx.i < ctx.src.length && /[a-zA-Z0-9_]/.test(ctx.src[ctx.i])) {
        id += ctx.src[ctx.i];
        ctx.advance();
    }
    ctx.pushToken({ type: KEYWORDS.has(id) ? 'KEYWORD' : 'IDENTIFICATOR', value: id, line: startLine, col: startCol });
}

function handleString(ctx, startLine, startCol, errors) {
    const quote = ctx.src[ctx.i];
    ctx.advance();
    let str = '';

    while (ctx.i < ctx.src.length && ctx.src[ctx.i] !== quote) {
        if (ctx.src[ctx.i] === '\n') {
            errors.push(new Error(`String not closed (line ${startLine}, col ${startCol})`));
        }
        str += ctx.src[ctx.i];
        ctx.advance();
    }

    if (ctx.i >= ctx.src.length) {
        errors.push(new Error(`String not closed (line ${startLine}, col ${startCol})`));
        return;
    }

    ctx.advance();
    ctx.pushToken({ type: 'STRING', value: str, line: startLine, col: startCol });
}

function tokenize(src) {
    const tokens = [];
    const errors = [];
    const ctx = {
        src,
        i: 0,
        line: 1,
        col: 1,
        advance() {
            if (this.src[this.i] === '\n') { this.line++; this.col = 1; }
            else this.col++;
            this.i++;
        },
        tokenStart: 0,
        // start/end = offsets in the source, end = position right after token
        pushToken(t) { tokens.push({ ...t, start: this.tokenStart, end: this.i }); }
    };

    while (ctx.i < ctx.src.length) {
        const c = ctx.src[ctx.i];
        const startLine = ctx.line, startCol = ctx.col;
        ctx.tokenStart = ctx.i;

        if (/\s/.test(c)) { ctx.advance(); continue; }

        if (c === '/' && ctx.src[ctx.i + 1] === '/') {
            while (ctx.i < ctx.src.length && ctx.src[ctx.i] !== '\n') ctx.advance();
            continue;
        }

        if (c === '-' && ctx.src[ctx.i + 1] === '>') {
            ctx.advance();
            ctx.advance();
            ctx.pushToken({ type: 'ARROW', value: '->', line: startLine, col: startCol });
            continue;
        }

        if (PUNCTUATION[c]) {
            ctx.advance();
            ctx.pushToken({ type: PUNCTUATION[c], value: c, line: startLine, col: startCol });
            continue;
        }

        if (c === '-' && /[0-9]/.test(ctx.src[ctx.i + 1])) {
            handleNumbers(ctx, startLine, startCol);
            continue;
        }

        if (/[0-9]/.test(c)) {
            handleNumbers(ctx, startLine, startCol);
            continue;
        }

        if (/[a-zA-Z_]/.test(c)) {
            handleLetters(ctx, startLine, startCol);
            continue;
        }

        if (c === '"' || c === "'") {
            handleString(ctx, startLine, startCol, errors);
            continue;
        }

        errors.push(new Error(`Unknown character '${c}' (line ${ctx.line}, col ${ctx.col})`));
        ctx.advance();
    }

    ctx.tokenStart = ctx.i;
    ctx.pushToken({ type: 'EOF', value: null, line: ctx.line, col: ctx.col });
    return { tokens, errors };
}

function parseTokens(tokens) {
    let pos = 0;

    const peek = () => tokens[pos] || { type: 'EOF', value: null, line: 0, col: 0 };
    const advance = () => tokens[pos++];
    const expect = (type) => {
        const t = peek();
        if (t.type !== type) {
            const error = new Error(`Expected ${type} but got ${t.type} (line ${t.line}, col ${t.col})`);
            error.line = t.line;
            error.col = t.col;
            throw error;
        }
        return advance();
    };

    // program = statement* EOF
    function parseProgram() {
        const statements = [];
        while (peek().type !== 'EOF') {
            statements.push(parseStatement());
        }
        return { type: 'program', statements };
    }

    // statement = arc ";" | declaration ";"
    function parseStatement() {
        const t = peek();
        const isArc = t.type === 'IDENTIFICATOR' && tokens[pos + 1] && tokens[pos + 1].type === 'ARROW';

        const statement = isArc ? parseArc() : parseDeclaration();
        const semi = expect('SEMI');
        return { ...statement, start: t.start, end: semi.end };
    }

    function parseArc() {
        const fromToken = expect('IDENTIFICATOR');
        expect('ARROW');
        const toToken = expect('IDENTIFICATOR');
        const list = peek().type === 'LPAREN' ? parseAttributes() : null;
        return {
            type: 'arc',
            from: fromToken.value,
            to: toToken.value,
            attributes: list?.attributes ?? [],
            attributeList: list && { start: list.start, end: list.end },
            headerEnd: toToken.end,
            line: fromToken.line,
            col: fromToken.col,
            fromPos: { line: fromToken.line, col: fromToken.col },
            toPos: { line: toToken.line, col: toToken.col },
        };
    }

    // declaration = KEYWORD IDENT attributes? | IDENT attributes
    function parseDeclaration() {
        const t = peek();

        if (t.type === 'KEYWORD') {
            advance();
            const nameToken = expect('IDENTIFICATOR');
            const list = peek().type === 'LPAREN' ? parseAttributes() : null;
            return {
                type: 'declaration',
                keyword: t.value,
                name: nameToken.value,
                attributes: list?.attributes ?? [],
                attributeList: list && { start: list.start, end: list.end },
                headerEnd: nameToken.end,
                line: t.line,
                col: t.col,
                namePos: { line: nameToken.line, col: nameToken.col },
            };
        }

        if (t.type === 'IDENTIFICATOR') {
            const nameToken = advance();
            const list = parseAttributes();
            return {
                type: 'configuration',
                name: nameToken.value,
                attributes: list.attributes,
                attributeList: { start: list.start, end: list.end },
                headerEnd: nameToken.end,
                line: t.line,
                col: t.col,
                namePos: { line: nameToken.line, col: nameToken.col },
            };
        }

        const error = new Error(`Expected declaration but got ${t.type} (line ${t.line}, col ${t.col})`);
        error.line = t.line;
        error.col = t.col;
        throw error;
    }

    // attributes = "(" attribute ("," attribute)* ")"
    function parseAttributes() {
        const lparen = expect('LPAREN');
        const attributes = [];

        if (peek().type !== 'RPAREN') {
            attributes.push(parseAttribute());
            while (peek().type === 'COMMA') {
                advance();
                attributes.push(parseAttribute());
            }
        }

        const rparen = expect('RPAREN');
        return { attributes, start: lparen.start, end: rparen.end };
    }

    // attribute = IDENT ":" value
    function parseAttribute() {
        const keyToken = expect('IDENTIFICATOR');
        expect('COLON');
        const valueToken = parseValue();
        return {
            key: keyToken.value,
            value: valueToken.value,
            line: keyToken.line,
            col: keyToken.col,
            start: keyToken.start,
            end: valueToken.end,
            valueStart: valueToken.start,
        };
    }

    // value = STRING | NUMBER
    function parseValue() {
        const t = peek();
        if (t.type === 'STRING' || t.type === 'NUMBER') return advance();
        const error = new Error(`Expected value (STRING or NUMBER) but got ${t.type} (line ${t.line}, col ${t.col})`);
        error.line = t.line;
        error.col = t.col;
        throw error;
    }

    return parseProgram();
}

 * @param {'place'|'transition'|'arc'} kind
 * @param {string} key - Canonical attribute key.
 * @returns {string}

 * Resolves what kind of node a statement describes.
 * @param {Object} statement - AST node.
 * @param {Map<string, Object>} symbols - Declared nodes by name.
 * @returns {string|undefined} 'place' | 'transition' | 'arc' | undefined.

    if (statement.type === 'declaration') return statement.keyword;

    number:         { test: (v) => typeof v === 'number',         label: 'a number' },
    string:         { test: (v) => typeof v === 'string',         label: 'a string' },
    nonNegativeInt: { test: (v) => Number.isInteger(v) && v >= 0, label: 'a whole number ≥ 0' },
    positiveInt:    { test: (v) => Number.isInteger(v) && v >= 1, label: 'a whole number ≥ 1' },

}

/**
 * Builds derived views of the AST shared by all rules.
 * @param {Object} model - The parsed model containing statements.
 * @returns {Object} Context.
 */
function buildContext(model) {
    const { statements } = model;
    const declarations = statements.filter((s) => s.type === 'declaration');

    return {
        statements,
        declarations,
        arcs: statements.filter((s) => s.type === 'arc'),
        configurations: statements.filter((s) => s.type === 'configuration'),
        symbols: new Map(declarations.map((d) => [d.name, d])),
    };
}

/**
 * Creates a diagnostic in the shape shared by parser and semantics.
 * @param {'error'|'warning'|'info'} severity
 * @param {string} message
 * @param {{line: number, col: number}} pos - Any AST node or position object.
 * @returns {Object} Diagnostic.
 */
function diagnostic(severity, message, pos) {
    return {
        severity,
        message,
        line: pos.line,
        col: pos.col,
    };
}

/**
 * Returns every repeated occurrence of a key together with its first occurrence.
 * @param {Array} items - Items to check.
 * @param {Function} getKey - Returns the comparison key for an item.
 * @returns {Array<{item: *, first: *, key: string}>} Duplicates.
 */
function findDuplicates(items, getKey) {
    const seen = new Map();
    const duplicates = [];

    for (const item of items) {
        const key = getKey(item);
        const first = seen.get(key);

        if (first) duplicates.push({ item, first, key });
        else seen.set(key, item);
    }

    return duplicates;
}

/**

 * @param {Object} statement - AST node.
 * @returns {string}

function labelOf(statement) {
    return statement.type === 'arc' ? `${statement.from}->${statement.to}` : statement.name;
}

/**
 * Pairs every attribute with the schema entry it should match.
 * @param {Object} ctx - Context from buildContext.
 * @returns {Array<{attr: Object, kind: string, key: string, expected: *, schema: Object}>}
 *          expected is undefined when the attribute is not in the schema for that kind.
 */
function schemaEntries({ statements, symbols }) {
    return statements.flatMap((statement) => {
        const kind = kindOf(statement, symbols);
        const schema = ATTRIBUTE_SCHEMA[kind];
        if (!schema) return [];

        return statement.attributes.map((attr) => {
            const key = canonicalKey(attr.key);
            const expected = Object.hasOwn(schema, key) ? schema[key] : undefined;
            return { attr, kind, key, expected, schema };
        });
    });
}

/**
 * Finds a schema key close enough to be a typo (makrings -> markings).
 * @param {string} key - Unknown key.
 * @param {Object} schema - Schema for the node kind.
 * @returns {string|undefined} Suggested key.
 */
function suggestKey(key, schema) {
    let best;
    let bestDistance = Infinity;

    for (const candidate of Object.keys(schema)) {
        const distance = levenshtein(key, candidate);
        if (distance < bestDistance) {
            best = candidate;
            bestDistance = distance;
        }
    }

    return best && bestDistance <= 2 && bestDistance < best.length / 2 ? best : undefined;
}

/** 
 * Computes the Levenshtein distance between two strings.
 * @param {string} a - First string.
 * @param {string} b - Second string.
 * @returns {number} Levenshtein distance.
 */
function levenshtein(a, b) {
    let prev = Array.from({ length: b.length + 1 }, (_, i) => i);

    for (let i = 1; i <= a.length; i++) {
        const curr = [i];
        for (let j = 1; j <= b.length; j++) {
            const cost = a[i - 1] === b[j - 1] ? 0 : 1;
            curr[j] = Math.min(prev[j] + 1, curr[j - 1] + 1, prev[j - 1] + cost);
        }
        prev = curr;
    }

    return prev[b.length];
}

/**
 * place p1; place p1;
 */
function checkUniqueNames({ declarations }) {
    return findDuplicates(declarations, (decl) => decl.name).map(({ item, first }) =>
        diagnostic('error', `'${item.name}' is already declared on line ${first.line}`, item.namePos)
    );
}

/**
 * p1 -> t9; p9 (x: 10); (t9, p9 not declared)
 */
function checkUndeclaredNames({ arcs, configurations, symbols }) {
    const references = [
        ...arcs.flatMap((arc) => [
            { name: arc.from, pos: arc.fromPos },
            { name: arc.to, pos: arc.toPos },
        ]),
        ...configurations.map((config) => ({ name: config.name, pos: config.namePos })),
    ];

    return references
        .filter(({ name }) => !symbols.has(name))
        .map(({ name, pos }) => diagnostic('error', `'${name}' is not declared`, pos));
}

/**

function checkArcEndpoints({ arcs, symbols }) {

        .filter(({ from, to }) => from && to && from.keyword === to.keyword)

 * inhibitor/reset/read : place -> transition
 */
function checkSpecialArcDirection({ arcs, symbols }) {
    return arcs.flatMap((arc) => {
        const typeAttr = arc.attributes.find((attr) => canonicalKey(attr.key) === 'type');
        if (!typeAttr || !PLACE_TO_TRANSITION_ONLY.has(typeAttr.value)) return [];

        const from = symbols.get(arc.from);
        if (!from || from.keyword === 'place') return [];

        return [diagnostic('error', `Arc of type '${typeAttr.value}' must go from a place to a transition`, typeAttr)];
    });
}

/**
 * p1 -> t1; p1 -> t1;
 */
function checkUniqueArcs({ arcs }) {
    return findDuplicates(arcs, (arc) => labelOf(arc)).map(({ item, first, key }) =>
        diagnostic('warning', `Arc '${key}' is already declared on line ${first.line}, ignoring this one`, item)
    );
}

/**
 * place p1 (m: 2, tokens: 3); p1 (x: 10); p1 (x: 20); p1 -> t1 (w: 1, w: 2);
 */
function checkUniqueAttributes({ statements }) {
    const entries = statements.flatMap((statement) =>
        statement.attributes.map((attr) => ({ owner: labelOf(statement), attr }))
    );

    return findDuplicates(entries, (entry) => `${entry.owner}.${canonicalKey(entry.attr.key)}`).map(({ item, first }) =>
        diagnostic(
            'warning',
            `'${item.attr.key}' on '${item.owner}' is already set as '${first.attr.key}' on line ${first.attr.line}, ignoring this one`,
            item.attr
        )
    );
}

/**
 * place p1 (makrings: 2); 
 * - results in error with suggestion
 * place p1 (color: "red"); 
 * - results error
 */
function checkUnknownAttributes(ctx) {
    return schemaEntries(ctx)
        .filter(({ expected }) => expected === undefined)
        .map(({ attr, kind, key, schema }) => {
            const suggestion = suggestKey(key, schema);
            const hint = suggestion ? `, did you mean '${suggestion}'?` : '';

            return diagnostic('error', `Unknown attribute '${attr.key}' on ${kind}${hint}`, attr);
        });
}

/**
 * place p1 (tokens: "abc"); p1 -> t1 (weight: 0); p1 -> t1 (type: "unknown");
 */
function checkAttributeValues(ctx) {
    return schemaEntries(ctx)
        .filter(({ attr, expected }) => expected !== undefined && !matchesSchema(attr.value, expected))
        .map(({ attr, expected }) =>
            diagnostic('error', `'${attr.key}' must be ${describeExpected(expected)}`, attr)
        );
}

/**
 * place lonely;   
 * no arc connects to object.
 */
function checkUnconnectedNodes({ declarations, arcs }) {
    const connected = new Set(arcs.flatMap((arc) => [arc.from, arc.to]));

    return declarations
        .filter((decl) => !connected.has(decl.name))
        .map((decl) => diagnostic('info', `${decl.keyword} '${decl.name}' is not connected`, decl.namePos));
}

===== src/core/petriEngine.js
/**
 * Checks whether a transition is enabled under a given marking.
 * A transition with no input places is always enabled (source transition).
 *
 * @param {Object} net - The network structure ({ arcs }).
 * @param {Object<string, number>} marking - Token count per place ID.
 * @param {string} transitionId - The ID of the transition to check.
 * @returns {boolean} True if the transition is enabled.
 */
export const isEnabled = (net, marking, transitionId) =>
  net.arcs
    .filter((arc) => arc.target === transitionId)
    .every((arc) => (marking[arc.source] ?? 0) >= arc.weight);

/**
 * Fires a transition, returning the resulting marking.
 * Returns null if the transition is not enabled.
 *
 * @param {Object} net - The network structure ({ arcs }).
 * @param {Object<string, number>} marking - The current marking.
 * @param {string} transitionId - The ID of the transition to fire.
 * @returns {Object<string, number>|null} The new marking, or null.
 */
export const fire = (net, marking, transitionId) => {
  if (!isEnabled(net, marking, transitionId)) return null;

  const next = { ...marking };

  for (const arc of net.arcs) {
    if (arc.target === transitionId) next[arc.source] -= arc.weight;
    if (arc.source === transitionId) next[arc.target] += arc.weight;
  }

  return next;
};

/**
 * Returns the IDs of all transitions enabled under the given marking.
 *
 * @param {Object} net - The network structure ({ transitions, arcs }).
 * @param {Object<string, number>} marking - The current marking.
 * @returns {Array<string>} IDs of enabled transitions.
 */
export const getEnabledTransitions = (net, marking) =>
  net.transitions.filter((t) => isEnabled(net, marking, t.id)).map((t) => t.id);

/**
 * Builds the initial marking from the places tokens.
 *
 * @param {Array<Object>} places - The list of places.
 * @returns {Object<string, number>} The initial marking.
 */
export const buildInitialMarking = (places) =>
  Object.fromEntries(places.map((p) => [p.id, p.tokens ?? 0]));

/**
 * Validates whether an arc can be created between a source and a target node (bipartite graph).
 *
 * @param {string} sourceId - The ID of the source node.
 * @param {string} targetId - The ID of the target node.
 * @param {Array<Object>} places - The list of all places in the network.
 * @param {Array<Object>} transitions - The list of all transitions in the network.
 * @returns {boolean} True if the connection is bipartite-valid, false otherwise.
 */
export const isValidArc = (sourceId, targetId, places, transitions) => {
  if (sourceId === targetId) return false;

  const isSourcePlace = places.some((p) => p.id === sourceId);
  const isTargetPlace = places.some((p) => p.id === targetId);

  const isSourceTransition = transitions.some((t) => t.id === sourceId);
  const isTargetTransition = transitions.some((t) => t.id === targetId);

  return (isSourcePlace && isTargetTransition) || (isSourceTransition && isTargetPlace);
};

import { CANVAS_CONFIG } from '../../../constants/layout';
import { canvasToCell } from '../../../utils/grid';

/**
 * Calculates the target zoom level based on the current zoom and the wheel delta.
 * @param {number} currentZoom - The current zoom level.
 * @param {number} deltaY - The wheel delta (positive for zoom out, negative for zoom in).
 * @returns {number} - Calculated target zoom level.
 */
export const calculateTargetZoom = (currentZoom, deltaY) => {
    const direction = deltaY < 0 ? 1 : -1;
    
    return direction > 0 
        ? Math.min(currentZoom * CANVAS_CONFIG.ZOOM.FACTOR, CANVAS_CONFIG.ZOOM.MAX) 
        : Math.max(currentZoom / CANVAS_CONFIG.ZOOM.FACTOR, CANVAS_CONFIG.ZOOM.MIN);
};

/**
 * Calculates the new pan position so that the point under the mouse remains in the same place after a zoom change.
 * @param {Object} params - Parameters for the calculation.
 * @param {number} params.mouseX - X-coordinate of mouse position.
 * @param {number} params.mouseY - Y-coordinate of mouse position.
 * @param {Object} params.currentPan - Current pan position.
 * @param {number} params.currentZoom - Current zoom level.
 * @param {number} params.targetZoom - Target zoom level.
 * @returns {Object} - Calculated new pan position.
 */
export const calculateZoomPan = ({ mouseX, mouseY, currentPan, currentZoom, targetZoom }) => {
    const canvasX = (mouseX - currentPan.x) / currentZoom;
    const canvasY = (mouseY - currentPan.y) / currentZoom;

    return {
        x: mouseX - canvasX * targetZoom,
        y: mouseY - canvasY * targetZoom,
    };
};

/**
 * Calculates the canvas coordinates based on the screen coordinates.
 * @param {number} clientX - The x-coordinate of the mouse position on the screen.
 * @param {number} clientY - The y-coordinate of the mouse position on the screen.
 * @param {HTMLElement} svgElement - The SVG element.
 * @param {Object} pan - Current pan position.
 * @param {number} zoom - Current zoom level.
 * @returns {Object} - Calculated canvas coords.
 */
export const screenToCanvasCoordinates = (clientX, clientY, svgElement, pan, zoom) => {
    if (!svgElement) return { x: 0, y: 0 };

    const rect = svgElement.getBoundingClientRect();
    return screenToCanvas(
        {
            x: clientX - rect.left,
            y: clientY - rect.top,
        },
        pan,
        zoom
    );
};

/**
 * Calculates the initial pan position based on height of the SVG element.
 * @param {number} height - The height of the SVG element.
 * @returns {Object} - Calculated initial pan position.
 */
export const calculateInitialPan = (height) => {
    return {
        x: CANVAS_CONFIG.ORIGIN_MARGIN_PX,
        y: height / 2 + CANVAS_CONFIG.ORIGIN_MARGIN_PX,
    };
};

/**
 * Converts canvas coordinates to screen coordinates.
 * @param {Object} point - Canvas coords.
 * @param {number} point.x - x-coord.
 * @param {number} point.y - y-coord.
 * @param {Object} pan - Current pan position.
 * @param {number} zoom - Current zoom level.
 * @returns {Object} - Screen coords (pixels in svg).
 */
export const canvasToScreen = ({ x, y }, pan, zoom) => {
    return {
        x: x * zoom + pan.x,
        y: y * zoom + pan.y,
    };
};

/**
 * Converts screen coordinates to canvas coordinates.
 * @param {Object} point - Screen coords.
 * @param {number} point.x - x-coord.
 * @param {number} point.y - y-coord.
 * @param {Object} pan - Current pan position.
 * @param {number} zoom - Current zoom level.
 * @returns {Object} - Canvas coords.
 */
export const screenToCanvas = ({ x, y }, pan, zoom) => {
    return {
        x: (x - pan.x) / zoom,
        y: (y - pan.y) / zoom,
    };
};

/**
 * Computes the logical grid cell bounds currently visible in the viewport.
 * Converts the screen pixel bounds to canvas coords, then to cell coords (only for cells without rulers).
 * @param {number} width - Viewport width in screen pixels.
 * @param {number} height - Viewport height in screen pixels.
 * @param {Object} pan - Current pan offset.
 * @param {number} zoom - Current zoom level.
 * @returns {{ minX: number, maxX: number, minY: number, maxY: number }}
 */
export const getVisibleCells = (width, height, pan, zoom) => {
    const rulerThickness = CANVAS_CONFIG.RULER_THICKNESS_PX;
    
    const topLeft = canvasToCell(screenToCanvas({ x: 0, y: 0 }, pan, zoom));
    const bottomRight = canvasToCell(screenToCanvas({ x: width - rulerThickness, y: height - rulerThickness }, pan, zoom));

    const minX = topLeft.x;
    const maxX = bottomRight.x;
    const minY = bottomRight.y;
    const maxY = topLeft.y;

    return { minX, maxX, minY, maxY };
};

import { TOOLS, MOUSE_BUTTON, CANVAS_ACTION } from '../../constants/tools';

    shouldStartPanning,
    isCanvasBackgroundClick,
    getElementFromEvent,
    resolveBackgroundClickAction,
    resolveContextMenuAction,
} from './helpers/canvasInteractions';
import { canvasToCell } from '../../utils/grid';

/**
 * Mouse handling on the canvas: adding nodes, selection, deleting and dragging nodes.
 */
export const useCanvasInteractions = ({
    selectedTool,
    addPlace,
    addTransition,
    moveNode,
    getCanvasCoordinates,
    startPanning,
    setSelectedTool,
    cancelConnecting,
    select,
    deleteElement,
}) => {
    const [dragPosition, setDragPosition] = useState(null);
    const [draggingNodeId, setDraggingNodeId] = useState(null);

    const handleMouseDown = (e) => {
        if (e.button === MOUSE_BUTTON.RIGHT) return;

        if (shouldStartPanning(e.button, e.shiftKey, selectedTool)) {
            startPanning(e.clientX, e.clientY);
            return;
        }

        if (!isCanvasBackgroundClick(e.target)) return;

        const { x, y } = canvasToCell(getCanvasCoordinates(e.clientX, e.clientY));
        const action = resolveBackgroundClickAction(selectedTool);

        switch (action.type) {
            case CANVAS_ACTION.ADD_PLACE:
                addPlace(x, y);
                break;
            case CANVAS_ACTION.ADD_TRANSITION:
                addTransition(x, y);
                break;
            case CANVAS_ACTION.CLEAR_SELECTION:
                select(null);
                break;
            default:
                break;
        }
    };

    const handleContextMenu = (e) => {
        e.preventDefault();

        const element = getElementFromEvent(e.target);
        const action = resolveContextMenuAction(selectedTool, element);

        switch (action.type) {
            case CANVAS_ACTION.RESET_TOOL:
                cancelConnecting?.();
                setSelectedTool(TOOLS.SELECT);
                break;
            case CANVAS_ACTION.DELETE_ELEMENT:
                deleteElement(action.id);
                break;
            default:
                break;
        }
    };

    const startDraggingNode = (nodeId) => {
        if (selectedTool !== TOOLS.SELECT) return;
        setDraggingNodeId(nodeId);
    };

    const handleMouseMove = (clientX, clientY) => {
        if (!draggingNodeId) return;
        const { x, y } = canvasToCell(getCanvasCoordinates(clientX, clientY));
        setDragPosition({ id: draggingNodeId, x, y });
    };

    const stopDraggingNode = () => {
        if (dragPosition) moveNode(dragPosition.id, dragPosition.x, dragPosition.y);
        setDraggingNodeId(null);
        setDragPosition(null);
    };

    return {
        dragPosition,
        handleMouseDown,
        handleContextMenu,
        handleMouseMove,
        startDraggingNode,
        stopDraggingNode,
    };
};

import { useState, useRef, useEffect, useLayoutEffect } from 'react';
import { CANVAS_CONFIG } from '../../constants/layout';
import {
    calculateInitialPan,
    calculateTargetZoom,
    calculateZoomPan,
    screenToCanvasCoordinates,
} from './helpers/canvasPanZoom';

export const useCanvasPanZoom = () => {
    const [zoom, setZoom] = useState(CANVAS_CONFIG.ZOOM.DEFAULT);
    const [pan, setPan] = useState({ x: 0, y: 0 });
    const [isPanning, setIsPanning] = useState(false);
    const [panStart, setPanStart] = useState({ x: 0, y: 0 });

    const svgRef = useRef(null);
    const zoomRef = useRef(zoom);
    const panRef = useRef(pan);

    useEffect(() => { zoomRef.current = zoom; }, [zoom]);
    useEffect(() => { panRef.current = pan; }, [pan]);

    useLayoutEffect(() => {
        const svgElement = svgRef.current;
        if (!svgElement) return;

        const height = svgElement.getBoundingClientRect().height;
        if (height <= 0) return;

        setPan(calculateInitialPan(height));
    }, []);

    useEffect(() => {
        const svgElement = svgRef.current;
        if (!svgElement) return;

        const handleWheel = (e) => {
            e.preventDefault();

            const rect = svgElement.getBoundingClientRect();
            const mouseX = e.clientX - rect.left;
            const mouseY = e.clientY - rect.top;
            const currentZoom = zoomRef.current;
            const currentPan = panRef.current;
            const targetZoom = calculateTargetZoom(currentZoom, e.deltaY);

            if (targetZoom === currentZoom) return;

            const nextPan = calculateZoomPan({
                mouseX,
                mouseY,
                currentPan,
                currentZoom,
                targetZoom,
            });

            setPan(nextPan);
            setZoom(targetZoom);
        };

        svgElement.addEventListener('wheel', handleWheel, { passive: false });
        return () => svgElement.removeEventListener('wheel', handleWheel);
    }, []);

    const getCanvasCoordinates = (clientX, clientY) => {
        return screenToCanvasCoordinates(clientX, clientY, svgRef.current, pan, zoom);
    };

    const startPanning = (clientX, clientY) => {
        setIsPanning(true);
        setPanStart({ x: clientX - pan.x, y: clientY - pan.y });
    };

    const updatePanning = (clientX, clientY) => {
        if (isPanning) {
            setPan({ x: clientX - panStart.x, y: clientY - panStart.y });
        }
    };

    const stopPanning = () => {
        setIsPanning(false);
    };

    return {
        svgRef,
        zoom,
        pan,
        isPanning,
        getCanvasCoordinates,
        startPanning,
        updatePanning,
        stopPanning,
    };
};

import { StreamLanguage } from '@codemirror/language';
import { KEYWORDS } from '../../../constants/dpn.js';

/** 
 * Defines the DPN language for CodeMirror using StreamLanguage.
 * @returns {StreamLanguage} CodeMirror language extension for DPN.
 */
export const dpnLanguage = StreamLanguage.define({
    name: 'dpn',
    token(stream) {
        if (stream.eatSpace()) return null;

        if (stream.match('//')) {
            stream.skipToEnd();
            return 'comment';
        }
        if (stream.match('->')) return 'operator';
        if (stream.match(/^"(?:\\.|[^"])*"?/) || stream.match(/^'(?:\\.|[^'])*'?/)) return 'string';
        if (stream.match(/^-?\d+/)) return 'number';
        if (stream.match(/^[()]/)) return 'paren';
        if (stream.match(/^[:,;]/)) return 'punctuation';

        if (stream.match(/^[a-zA-Z_][a-zA-Z0-9_]*/)) {
            const word = stream.current();
            if (KEYWORDS.has(word)) return 'keyword';
            if (stream.match(/^\s*:/, false)) return 'propertyName';
            return 'variableName';
        }

        stream.next();
        return 'invalid';
    },
    languageData: {
        commentTokens: { line: '//' },
    },
});

import { useEffect, useRef } from 'react';
import {
    EditorView,
    keymap,
    lineNumbers,
    highlightActiveLine,
    highlightActiveLineGutter,
    drawSelection,
} from '@codemirror/view';
import { EditorState, Compartment } from '@codemirror/state';
import { defaultKeymap, history, historyKeymap, indentWithTab } from '@codemirror/commands';
import { bracketMatching } from '@codemirror/language';
import { lintGutter } from '@codemirror/lint';

import { languageExtensions } from './helpers/languageExtensions';
import { editorTheme } from './helpers/editorTheme';

export const useCodeEditor = ({ code, setCode, language, theme, isDark }) => {
    const hostRef = useRef(null);
    const viewRef = useRef(null);

    const setCodeRef = useRef(setCode);
    setCodeRef.current = setCode;

    const compartmentsRef = useRef(null);
    if (!compartmentsRef.current) {
        compartmentsRef.current = { lang: new Compartment(), theme: new Compartment() };
    }

    useEffect(() => {
        const { lang, theme: themeComp } = compartmentsRef.current;

        const view = new EditorView({
            parent: hostRef.current,
            state: EditorState.create({
                doc: code ?? '',
                extensions: [
                    lineNumbers(),
                    highlightActiveLineGutter(),
                    highlightActiveLine(),
                    drawSelection(),
                    history(),
                    bracketMatching(),
                    lintGutter(),
                    keymap.of([...defaultKeymap, ...historyKeymap, indentWithTab]),
                    EditorView.contentAttributes.of({
                        spellcheck: 'false',
                        autocorrect: 'off',
                        autocapitalize: 'off',
                    }),
                    EditorView.updateListener.of((update) => {
                        if (update.docChanged) setCodeRef.current(update.state.doc.toString());
                    }),
                    lang.of(languageExtensions(language)),
                    themeComp.of(editorTheme(theme, isDark)),
                ],
            }),
        });

        viewRef.current = view;
        return () => {
            view.destroy();
            viewRef.current = null;
        };
    }, []);

    // Listen for code change from outside the editor.
    useEffect(() => {
        const view = viewRef.current;
        if (!view) return;
        const current = view.state.doc.toString();
        const next = code ?? '';
        if (next !== current) view.dispatch({ changes: minimalChange(current, next) });
    }, [code]);

    // Listen for language change
    useEffect(() => {
        viewRef.current?.dispatch({
            effects: compartmentsRef.current.lang.reconfigure(languageExtensions(language)),
        });
    }, [language]);

    // Listen for theme change
    useEffect(() => {
        viewRef.current?.dispatch({
            effects: compartmentsRef.current.theme.reconfigure(editorTheme(theme, isDark)),
        });
    }, [theme, isDark]);

    return { hostRef, viewRef };
};

/**
 * The smallest single replacement that turns before to after
 * (skips the common start and end of both texts).
 */
const minimalChange = (before, after) => {
    let start = 0;
    while (start < before.length && start < after.length && before[start] === after[start]) start++;

    let endBefore = before.length;
    let endAfter = after.length;
    while (endBefore > start && endAfter > start && before[endBefore - 1] === after[endAfter - 1]) {
        endBefore--;
        endAfter--;
    }

    return { from: start, to: endBefore, insert: after.slice(start, endAfter) };
};

import { useEffect, useState } from 'react';
import * as dpnParser from '../core/parsers/dpnParser';

const DEFAULT_DELAY_MS = 450;

function normalizeParseError(err) {
    if (!err) return { message: 'Unknown error' };
    if (typeof err === 'string') return { message: err };
    return {
        message: err.message || err.msg || String(err),
        line: err.line,
        col: err.col,
        severity: err.severity,
    };
}

export const useCodeChecker = ({ code = '', language = 'dpn', delayMs = DEFAULT_DELAY_MS } = {}) => {
    const [errors, setErrors] = useState([]);

    useEffect(() => {
        if (language !== 'dpn') {
            setErrors([]);
            return undefined;
        }

        const timer = setTimeout(() => {
            try {
                const parseFn = dpnParser.parse || (dpnParser.default && dpnParser.default.parse);
                if (!parseFn) {
                    setErrors([normalizeParseError('Parser not available')]);
                    return;
                }

                const parsed = parseFn(code) || {};
                const parsedErrors = Array.isArray(parsed.errors) ? parsed.errors : [];

                setErrors(parsedErrors.length > 0 ? parsedErrors : []);
            } catch (err) {
                setErrors([normalizeParseError(err)]);
            }
        }, delayMs);

        return () => clearTimeout(timer);
    }, [code, language, delayMs]);

    console.log('useCodeChecker errors:', errors);
    return errors;
};

export default useCodeChecker;

/**
 * True when the user is typing somewhere (input, textarea, CodeMirror...),
 * so single-key shortcuts must not fire.
 */
const isTypingTarget = (element) =>
    element instanceof HTMLElement &&
    (element.isContentEditable || ['INPUT', 'TEXTAREA', 'SELECT'].includes(element.tagName));

/**
 * Global single-key shortcuts.
 *
 * @param {Array<{
 *   codes: string[],            // KeyboardEvent.code values that trigger it
 *   handler: Function,          // what to do
 *   enabled?: boolean,          // false = key passes through untouched (default true)
 *   repeat?: boolean,           // true = fires again while the key is held (default false)
 * }>} bindings
 */
export const useKeyboardShortcuts = (bindings) => {
    const bindingsRef = useRef(bindings);
    bindingsRef.current = bindings;

    useEffect(() => {
        const handleKeyDown = (event) => {
            if (event.defaultPrevented || event.isComposing) return;
            if (event.ctrlKey || event.metaKey || event.altKey) return;
            if (isTypingTarget(event.target)) return;

            const binding = bindingsRef.current.find((b) => b.codes?.includes(event.code));
            if (!binding || binding.enabled === false) return;

            event.preventDefault();
            if (event.repeat && !binding.repeat) return;

            binding.handler(event);
        };

        window.addEventListener('keydown', handleKeyDown);
        return () => window.removeEventListener('keydown', handleKeyDown);
    }, []);
};

===== src/hooks/useSyncedScroll.js
/**
 * Keeps scroll containers at the same scroll position.
 * bind(index) for each container and pass the result to it.
 * @returns {(index: number) => { ref: Function, onScroll: Function }}
 */
import { useRef } from 'react';

export const useSyncedScroll = () => {
    const containers = useRef([]);

    const bind = (index) => ({
        ref: (element) => {
            containers.current[index] = element;
        },
        onScroll: (event) => {
            const { scrollTop, scrollLeft } = event.currentTarget;

            containers.current.forEach((element, i) => {
                if (i === index || !element) return;
                // Stops echoing the scroll event back to original container
                if (element.scrollTop !== scrollTop) element.scrollTop = scrollTop;
                if (element.scrollLeft !== scrollLeft) element.scrollLeft = scrollLeft;
            });
        },
    });

    return bind;
};

     * @param {import('../../core/models').Net} net

        set((state) => {
            if (state.mode === 'run') return {};

import { create } from 'zustand';
import { PANELS } from '../constants/workspace.js';
import { applyDefaultLayout, revealPanel } from '../utils/dockLayout.js';

/**
 * Workspace state store.
 */
export const useWorkspaceStore = create((set, get) => ({
    dockApi: null,
    mobilePanel: PANELS[0].id,

    setDockApi: (dockApi) => set({ dockApi }),

    /**
     * Shows a panel – focuses or reopens it on desktop, switches to it on mobile.
     * @param {string} id - Panel id from PANELS.
     */
    showPanel: (id) => {
        const { dockApi } = get();
        if (dockApi) revealPanel(dockApi, id);
        else set({ mobilePanel: id });
    },

    /** 
     * Restores to default. 
     */
    resetLayout: () => {
        const { dockApi } = get();
        if (dockApi) applyDefaultLayout(dockApi);
        else set({ mobilePanel: PANELS[0].id });
    },
}));

===== src/utils/array.js
/**
 * Includes <from, to> in the range.
 * @param {number} from - Start
 * @param {number} to - End
 * @returns {number[]} - The range of numbers from `from` to `to`
 */
export function range(from, to) {
    const result = [];

    for (let i = from; i <= to; i++) {
        result.push(i);
    }

    return result;
}

/**
 * Creates a matrix filled with 0s.
 * @param {number} rows 
 * @param {number} cols 
 * @returns {number[][]} - Matrix filled with 0s.
 */
export const zeros = (rows, cols) => {
    const matrix = [];

    for (let r = 0; r < rows; r++) {
        const row = [];
        for (let c = 0; c < cols; c++) {
            row.push(0);
        }
        matrix.push(row);
    }

    return matrix;
};
===== src/utils/dockLayout.js
import { PANELS, DEFAULT_LAYOUT } from '../constants/workspace.js';

const titleOf = (id) => PANELS.find((panel) => panel.id === id)?.title ?? id;

/**
 * Adds a panel where DEFAULT_LAYOUT says.
 * @param {import('dockview-react').DockviewApi} api
 * @param {string} id - Panel id from PANELS.
 */
const addPanel = (api, id) => {
    const { position, initialWidth, initialHeight } = DEFAULT_LAYOUT.find((entry) => entry.id === id) ?? {};
    const hasReference = position && api.getPanel(position.referencePanel);

    return api.addPanel({
        id,
        component: id,
        title: titleOf(id),
        renderer: 'always', // keep the DOM when the tab is hidden (CodeMirror history, canvas zoom)
        ...(hasReference && { position }),
        ...(initialWidth && { initialWidth }),
        ...(initialHeight && { initialHeight }),
    });
};

/**
 * Applies default layout.
 * @param {import('dockview-react').DockviewApi} api
 */
export const applyDefaultLayout = (api) => {
    api.clear();
    DEFAULT_LAYOUT.forEach(({ id }) => addPanel(api, id));
    api.getPanel(DEFAULT_LAYOUT[0].id)?.api.setActive();
};

/**
 * Reveals a panel - focuses or reopens it.
 * @param {import('dockview-react').DockviewApi} api
 * @param {string} id - Panel id from PANELS.
 */
export const revealPanel = (api, id) => {
    const panel = api.getPanel(id);
    if (panel) panel.api.setActive();
    else addPanel(api, id);
};

===== src/utils/grid.js
import { CANVAS_CONFIG } from '../constants/layout';

const cellCenterOffset = CANVAS_CONFIG.GRID_SIZE_PX / 2;

/**
 * Converts a grid cell position to canvas coords.
 * counted upward from the origin.
 */
export function cellToCanvas({ x, y }) {
    return { x: x * CANVAS_CONFIG.GRID_SIZE_PX + cellCenterOffset, y: -(y * CANVAS_CONFIG.GRID_SIZE_PX + cellCenterOffset) };
}

/**
 * Converts canvas coords back to logical cell grid.
 * Y coord is flipped before flooring so values just above the
 * origin map to the correct cell, e.g. y = -1 -> cell y = 0.
 */
export function canvasToCell({ x, y }) {
    return { x: Math.floor(x / CANVAS_CONFIG.GRID_SIZE_PX), y: Math.floor((-y) / CANVAS_CONFIG.GRID_SIZE_PX)};
}
===== src/utils/keyboard.js
/**
 * Turns the first key code of a shortcut into the text shown in UI.
 * 'Digit3' -> '3', 'KeyC' -> 'C', 'Numpad3' -> 'Num 3'.
 * @param {string[]} [codes] - KeyboardEvent.code values.
 * @returns {string} Label, empty if no shortcut.
 */
export const shortcutLabel = (codes) => {
    const [code] = codes ?? [];
    if (!code) return '';
    return code.replace(/^(Digit|Key)/, '').replace(/^Numpad/, 'Num ');
};

/**
 * Appends the shortcut to tooltip text.
 * @param {string} label - Tooltip text.
 * @param {string[]} [codes] - KeyboardEvent.code values.
 * @returns {string}
 */
export const withShortcut = (label, codes) => {
    const shortcut = shortcutLabel(codes);
    return shortcut ? `${label} (${shortcut})` : label;
};

===== src/utils/time.js
const pad = (value) => String(value).padStart(2, '0');

/**
 * Formats a duration for the simulation timer.
 * 5 -> '00:05', 65 -> '01:05', 3665 -> '01:01:05', 90065 -> '01:01:01:05'
 * @param {number} totalSeconds - Seconds.
 * @returns {string} mm:ss | hh:mm:ss | dd:hh:mm:ss.
 */
export const formatElapsedTime = (totalSeconds) => {
    const days = Math.floor(totalSeconds / 86400);
    const hours = Math.floor(totalSeconds / 3600) % 24;
    const minutes = Math.floor(totalSeconds / 60) % 60;
    const seconds = totalSeconds % 60;

    const parts = days > 0
        ? [days, hours, minutes, seconds]
        : hours > 0
            ? [hours, minutes, seconds]
            : [minutes, seconds];

    return parts.map(pad).join(':');
};
